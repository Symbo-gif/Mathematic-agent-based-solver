stages:
  - security

strix-penetration-test:
  stage: security
  image: python:3.12-slim

  services:
    - name: docker:dind
      alias: docker

  variables:
    # Docker-in-Docker setup
    DOCKER_HOST: tcp://docker:2375
    DOCKER_TLS_CERTDIR: ""

    # Optional tweaks
    PIP_DISABLE_PIP_VERSION_CHECK: "1"
    PYTHONUNBUFFERED: "1"

    # Make sure you define STRIX_LLM, LLM_API_KEY, LLM_API_BASE, GITLAB_TOKEN
    # as GitLab CI/CD variables in the project or group.

  before_script:
    # Install Docker CLI so we can talk to the docker:dind service
    - apt-get update
    - apt-get install -y --no-install-recommends docker.io docker-cli
    - rm -rf /var/lib/apt/lists/*

    # Show docker binary & version for debugging (do NOT fail if daemon not ready yet)
    - which docker || echo "docker not found in PATH (before_script)"
    - docker version || echo "docker daemon not reachable yet (ignored here)"

    # Upgrade pip and install uv + requests into system Python
    - python -m pip install --upgrade pip
    - pip install uv requests

    # Create virtualenv for Strix
    - uv venv

  script:
    # Install Strix into the venv
    - uv pip install strix-agent

    # Quick docker sanity check right before Strix
    - which docker || echo "docker not found in PATH right before Strix"
    - docker version || echo "docker daemon not reachable (ignored here too)"

    # Run Strix (do NOT fail the job yet if Strix returns non-zero)
    - |
      set +e
      echo "Running Strix penetration test..."
      uv run strix -n -t ./pygoat -t http://vps.s3x.gdn:8998/
      STRIX_EXIT=$?
      echo "Strix exit code: ${STRIX_EXIT}"
      set -e

    # Create GitLab issues from Strix findings
    - |
      echo "Creating GitLab issues from Strix findings..."
      python - << 'PY'
      import os
      import csv
      import pathlib
      import sys
      import requests

      project_id = os.environ.get("CI_PROJECT_ID")
      if not project_id:
          print("CI_PROJECT_ID not set, aborting.", file=sys.stderr)
          sys.exit(0)

      api_url = os.environ.get("CI_API_V4_URL")
      if not api_url:
          server_url = os.environ.get("CI_SERVER_URL", "").rstrip("/")
          api_url = server_url + "/api/v4"

      token = os.environ.get("GITLAB_TOKEN")
      if not token:
          print("GITLAB_TOKEN is not set - cannot create issues, skipping.", file=sys.stderr)
          sys.exit(0)

      strix_root = pathlib.Path("strix_runs")
      if not strix_root.exists():
          print("No strix_runs directory - nothing to create.")
          sys.exit(0)

      csv_files = list(strix_root.rglob("vulnerabilities.csv"))
      if not csv_files:
          print("No vulnerabilities.csv files found under strix_runs.")
          sys.exit(0)

      print(f"Found {len(csv_files)} vulnerabilities.csv file(s).")

      session = requests.Session()
      session.headers["PRIVATE-TOKEN"] = token

      for csv_path in csv_files:
          base_dir = csv_path.parent
          print(f"Processing {csv_path}")
          with csv_path.open("r", encoding="utf-8") as f:
              reader = csv.DictReader(f)
              for row in reader:
                  vuln_id = row.get("id", "").strip()
                  title = row.get("title", "").strip()
                  severity = row.get("severity", "").strip()
                  timestamp = row.get("timestamp", "").strip()
                  file_rel = row.get("file", "").strip()

                  md_path = base_dir / file_rel
                  if md_path.exists():
                      vuln_body = md_path.read_text(encoding="utf-8")
                  else:
                      print(f"WARNING: markdown file not found for {vuln_id}: {md_path}", file=sys.stderr)
                      vuln_body = f"Could not read detailed vuln file at {md_path}"

                  issue_title = f"[STRIX][{severity}] {title}"
                  description = (
                      f"Strix vulnerability ID: `{vuln_id}`\n"
                      f"Severity: **{severity}**\n"
                      f"Timestamp: `{timestamp}`\n\n"
                      f"Source file: `{file_rel}`\n\n"
                      f"---\n\n"
                      f"{vuln_body}"
                  )

                  labels = ["security", "strix"]
                  if severity:
                      labels.append(severity.lower())

                  data = {
                      "title": issue_title,
                      "description": description,
                      "labels": ",".join(labels),
                  }

                  resp = session.post(f"{api_url}/projects/{project_id}/issues", data=data)
                  if resp.status_code >= 300:
                      print(
                          f"Failed to create issue for {vuln_id}: {resp.status_code} {resp.text}",
                          file=sys.stderr,
                      )
                  else:
                      iid = resp.json().get("iid")
                      print(f"Created issue !{iid} for {vuln_id}")
      PY

    # If you want Strix findings to fail the pipeline, uncomment:
    # - exit "${STRIX_EXIT}"

  artifacts:
    when: always
    paths:
      - strix_runs
